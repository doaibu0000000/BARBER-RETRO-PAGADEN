    43→/** Status buka/tutup berdasarkan jam WIB. Null sampai client mount (hindari mismatch). */
    44→function useOpenStatus() {
    45→  const [status, setStatus] = useState<null | boolean>(null);
    46→  useEffect(() => {
    47→    const compute = () => {
    48→      const fmt = new Intl.DateTimeFormat("en-GB", {
    49→        timeZone: "Asia/Jakarta",
    50→        hour: "2-digit",
    51→        minute: "2-digit",
    52→        hour12: false,
    53→      });
    54→      const [h, m] = fmt.format(new Date()).split(":").map(Number);
    55→      const minutes = h * 60 + m;
    56→      return minutes >= BUSINESS.openHour * 60 && minutes < BUSINESS.closeHour * 60;
    57→    };
    58→    const t = setInterval(() => setStatus(compute()), 60_000);
    59→    const raf = requestAnimationFrame(() => setStatus(compute()));
    60→    return () => {
    61→      clearInterval(t);
    62→      cancelAnimationFrame(raf);
    63→    };
    64→  }, []);
    65→  return status;
    66→}
